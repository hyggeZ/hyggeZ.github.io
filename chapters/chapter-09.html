协程

是用户级线程，在任何普通函数调用前加 go 关键字。因为主线程结束协程自动结束，所以需要等待协程

1. sync.WaitGroup 协程计数器

Go
var wg = sync.WaitGroup{}// 可以var wg sync.WaitGroup
wg.Add()  wg.Done()  wg.Wait()
Go
func worker(id int, wg *sync.WaitGroup) {
    defer wg.Done()
    fmt.Printf("工人 %d 开始干活...\n", id)
}

func main() {
    var wg sync.WaitGroup
    for i := 1; i <= 3; i++ {
        wg.Add(1)
        go worker(i, &wg)
    }
    wg.Wait()
    fmt.Println("所有工人干完活了，主程序安全退出。")
}

2. channel 通道

是一个线程安全的，自带阻塞锁和等待队列的物理管道

（1）创建管道，必须用make函数初始化

Go
ch := make(chan int)    // 创建一个只能传递 int 类型的管道
chStr := make(chan string) // 创建一个只能传递 string 类型的管道
ch := make(chan int, 3) // 创建一个容量为 3 的有缓冲通道

（2）发送数据 管道 <- 数据

Go
ch <- 42 // 把数字 42 推进 ch 管道

（3）接收数据 变量 <- 管道

Go
result := <-ch  // 从 ch 管道里捞出一个数据，赋值给 result
Go
func worker(id int, ch chan int, wg *sync.WaitGroup) {
    defer wg.Done()
    time.Sleep(time.Duration(id*100) * time.Millisecond)
    result := id*10 + 5
    ch <- result
    fmt.Printf("工人 %d：把结果 %d 扔进了通道\n", id, result)
}
Go
func main() {
    ch := make(chan int, 5)
    var wg sync.WaitGroup
    for i := 1; i <= 3; i++ {
        wg.Add(1)
        go worker(i, ch, &wg)
    }
    go func() {
        wg.Wait()
        close(ch)
    }()
    for val := range ch {
        fmt.Printf("成功捞出数据：%d\n", val)
    }
}

3. select 多路复用监听器

Go
func taskOne(ch1 chan string) {
    time.Sleep(2 * time.Second)
    ch1 <- "来自任务一的数据"
}
func taskTwo(ch2 chan string) {
    time.Sleep(1 * time.Second)
    ch2 <- "来自任务二的数据"
}

select {
    case res1 := <-ch1:
    fmt.Println("抓取到分支一：", res1)
    case res2 := <-ch2:
    fmt.Println("抓取到分支二：", res2)
}

select语句只执行一次，如果需要无限循环，要套一个for循环
